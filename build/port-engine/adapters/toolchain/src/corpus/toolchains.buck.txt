# The compiler is a hash-pinned download, not a PATH lookup, so it is a real input to
# every rustc action and no action shells out to rustup. The version is not configured
# here or in rust.bzl: it is read from rust-toolchain.toml via the `<file:>` include in
# .buckconfig, so there is one statement of it and this follows by construction.
rust_distribution(
    name = "rust_distribution",
    visibility = ["PUBLIC"],
)

hermetic_rust_toolchain(
    name = "rust",
    default_edition = "2024",
    distribution = ":rust_distribution",
    visibility = ["PUBLIC"],
)

# Absolute Unix compiler paths keep the prelude build-script cc-shim independent of PATH on
# Darwin/Linux. Windows must retain the prelude MSVC defaults; overriding them with Unix paths
# makes every Rust target that links a C build script fail before its tests can run.
CXX_TOOLCHAIN_CONFIG = {
    "compiler": None if host_info().os.is_windows else "/usr/bin/clang",
    "compiler_type": None if host_info().os.is_windows else "clang",
    "linker": None if host_info().os.is_windows else "/usr/bin/clang",
    "archiver": None if host_info().os.is_windows else "/usr/bin/ar",
}

system_cxx_toolchain(
    name = "cxx",
    visibility = ["PUBLIC"],
    **CXX_TOOLCHAIN_CONFIG
)

# REQUIRED by the prelude: prelude//decls/toolchains_common.bzl `_cxx_toolchain()`
# defaults to select({DEFAULT: toolchains//:cxx, config//:none: toolchains//:cxx_no_default_deps}).
# The config//:none branch is NEVER selected, but buck2 within-view/query validation
# requires the target to EXIST — without it, `//...` graph traversal (deps/rdeps, hence
# the affected-gate query) fails with "Unknown target cxx_no_default_deps". Mirror :cxx.
system_cxx_toolchain(
    name = "cxx_no_default_deps",
    visibility = ["PUBLIC"],
    **CXX_TOOLCHAIN_CONFIG
)

system_genrule_toolchain(
    name = "genrule",
    visibility = ["PUBLIC"],
)

# Required by prelude//rust/tools:internal_tools_toolchain
system_python_bootstrap_toolchain(

    visibility = ["PUBLIC"],

# Test runner toolchains — REQUIRED by `buck2 test`. The prelude resolves
# toolchains//:test (ExternalRunnerTestToolchainInfo, via noop_test_toolchain — the
# standard local runner the prelude demo wires) AND toolchains//:remote_test_execution
# (RemoteTestExecutionToolchainInfo). Without :test, any rust_test target fails to
# CONFIGURE ("Unknown target `test`"), so the gate's test phase can never run. Local
# execution (no RE configured) runs tests on the gate pod. Prerequisite for #84 (emit
# rust_test → gate builds AND TESTS). The RE target is wired now for when NativeLink RE lands.
noop_test_toolchain(
    name = "test",
    visibility = ["PUBLIC"],
)

remote_test_execution_toolchain(
    name = "remote_test_execution",
    visibility = ["PUBLIC"],
)
